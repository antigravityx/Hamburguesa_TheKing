rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // 1. CÁPSULAS DE ALMA (Perfil y Semilla del usuario)
    match /semillas/{userId} {
      // El Celador permite al usuario leer su propia cápsula o a todos si es un perfil público
      allow read: if true; 
      // Solo el dueño de la semilla puede modificar su ADN
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // 2. EL MURO (Red Social - Publicaciones)
    match /social_posts/{postId} {
      // Todos pueden ver el muro
      allow read: if true;
      // Solo los autenticados pueden publicar
      allow create: if request.auth != null;
      // Solo el autor puede borrar/editar
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.autor_uid;
      
      // 3. ECOS Y COMENTARIOS DE AUDIO
      match /comments/{commentId} {
        allow read: if true;
        allow create: if request.auth != null;
        allow update, delete: if request.auth != null && request.auth.uid == resource.data.autor_uid;
      }
    }
    
    // Reglas previas...
    match /{document=**} {
      allow read, write: if true; // (Cuidado: esto abre toda la base, idealmente deberías restringir las demás colecciones)
    }
  }
}
